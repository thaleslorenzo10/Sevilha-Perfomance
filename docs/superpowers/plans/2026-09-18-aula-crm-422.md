**Goal:** o webhook da Kiwify passa a avançar o deal `[AULA]` para "Pagou" mesmo quando a integração RD Marketing → CRM já o moveu para outro funil; quando não conseguir, deixa uma nota no deal e grava o motivo exato em `crm_status`.

**Architecture:** só `lib/kiwify.js` muda (função `avancarDeal`). O PUT passa a mandar `deal_pipeline_id` + `deal_stage_id` juntos; se o RD ainda recusar, `POST /activities` grava uma nota "Compra aprovada na Kiwify" no deal e o status devolve `erro:deal <status> <corpo>`. Um script avulso reprocessa as compras já gravadas com `crm_status` de erro.

**Tech Stack:** Node 24, `fetch` nativo, RD CRM API v1 (`https://crm.rdstation.com/api/v1`, `?token=`), Supabase REST. Teste = script avulso com `assert` e `global.fetch` mockado (padrão de `scripts/test-crm-webhook.js`).

```js
/**
 * Nota no deal quando o RD recusa a mudança de etapa: o vendedor vê a compra
 * mesmo com o deal parado no funil errado.
 */
async function anotarCompra(dealId, motivo) {
  const token = process.env.RD_CRM_TOKEN;
  const text = `Compra aprovada na Kiwify (${ROTULO}). O sistema não conseguiu mover para "Pagou": ${motivo}`;
  try {
    const r = await fetch(`${CRM_BASE}/activities?token=${token}`, {
      method: 'POST', headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ activity: { deal_id: dealId, text } }),
    });
    return r.ok;
  } catch (e) { console.warn('[kiwify] nota no deal falhou:', String(e.message).slice(0, 80)); return false; }
}

/**
 * Deal [AULA] do contato (pelo e-mail) → stage "pago".
 *
 * A integração RD Marketing → CRM move o deal para o funil da mentoria ~2 min
 * depois de criado (from_rdsm_integration). Um PUT só com deal_stage_id de
 * outro funil devolve 422 — por isso funil e etapa vão juntos. Se ainda assim
 * o RD recusar, fica uma nota no deal e o corpo da resposta vai no status.
 */
async function avancarDeal(email) {
  const token = process.env.RD_CRM_TOKEN, stage = process.env.RD_CRM_STAGE_ID_AULA_PAGO;
  const pipeline = process.env.RD_CRM_PIPELINE_ID_AULA;
  if (!token || !stage) return 'sem-config';
  const s = await fetch(`${CRM_BASE}/contacts?token=${token}&email=${encodeURIComponent(email)}`);
  if (!s.ok) return `erro:contato ${s.status}`;
  const contatos = (await s.json()).contacts || [];
  // A listagem devolve contatos "magros" (sem deal_ids) e o e-mail pode repetir em
  // mais de um contato — GET /contacts/:id é o único que traz o array completo.
  for (const contato of contatos) {
    const id = contato?._id || contato?.id;
    if (!id) continue;
    let dealIds = contato.deal_ids;
    const g = await fetch(`${CRM_BASE}/contacts/${id}?token=${token}`);
    if (g.ok) dealIds = (await g.json()).deal_ids;
    const ids = (dealIds || []).map(d => (typeof d === 'string' ? d : d._id || d.id)).filter(Boolean);
    for (const dealId of ids.reverse()) {
      const gd = await fetch(`${CRM_BASE}/deals/${dealId}?token=${token}`);
      if (!gd.ok) continue;
      const deal = await gd.json();
      if (!String(deal.name || '').startsWith('[AULA]')) continue;
      const corpo = { deal: pipeline ? { deal_pipeline_id: pipeline, deal_stage_id: stage } : { deal_stage_id: stage } };
      const p = await fetch(`${CRM_BASE}/deals/${dealId}?token=${token}`, {
        method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(corpo),
      });
      if (p.ok) return 'ok';
      const motivo = `${p.status} ${String(await p.text().catch(() => '')).replace(/\s+/g, ' ').slice(0, 80)}`.trim();
      await anotarCompra(dealId, motivo);
      return `erro:deal ${motivo}`;
    }
  }
  return 'sem-deal';
}
```

```js
#!/usr/bin/env node
'use strict';

/**
 * Reprocessa o RD CRM das compras da aula que ficaram com crm_status de erro
 * ou 'sem-deal' (ex.: as duas compras de 18/09/2026 que deram 'erro:deal 422').
 *
 *   node scripts/aula-reprocessar-crm.js            # lista o que faria
 *   node scripts/aula-reprocessar-crm.js --aplicar  # move os deals e grava o status
 *
 * Lê .env.local da raiz. Escreve no RD CRM e no Supabase só com --aplicar.
 */

const fs = require('fs'), path = require('path');
fs.readFileSync(path.join(__dirname, '..', '.env.local'), 'utf8')
  .split('\n').forEach(l => { const m = l.match(/^([A-Z_]+)=["']?([^"'\n]*)["']?$/); if (m && !process.env[m[1]]) process.env[m[1]] = m[2]; });

const { avancarDeal } = require('../lib/kiwify');
const { conexao, rest } = require('../lib/supabase');

const aplicar = process.argv.includes('--aplicar');
const TABELA = 'sevilha_compras_aula';

(async () => {
  const c = conexao();
  if (!c) { console.error('SUPABASE_URL / SUPABASE_SERVICE_KEY ausentes'); process.exit(1); }
  const r = await fetch(rest(TABELA, "status=eq.paid&or=(crm_status.like.erro*,crm_status.eq.sem-deal)&select=order_id,email,crm_status,pago_em&order=pago_em"), { headers: c.headers });
  if (!r.ok) { console.error('Supabase', r.status, await r.text()); process.exit(1); }
  const compras = await r.json();
  console.log(`${compras.length} compra(s) com CRM pendente${aplicar ? '' : ' — dry-run, use --aplicar para executar'}`);
  for (const compra of compras) {
    const mascara = compra.email.replace(/^(.{3}).*@/, '$1…@');
    if (!aplicar) { console.log(`  ${compra.order_id.slice(0, 8)}  ${mascara}  ${compra.crm_status}`); continue; }
    const status = await avancarDeal(compra.email);
    const p = await fetch(rest(TABELA, `order_id=eq.${encodeURIComponent(compra.order_id)}`), {
      method: 'PATCH', headers: { ...c.headers, Prefer: 'return=minimal' },
      body: JSON.stringify({ crm_status: status, updated_at: new Date().toISOString() }),
    });
    console.log(`  ${compra.order_id.slice(0, 8)}  ${mascara}  ${compra.crm_status} → ${status}${p.ok ? '' : ` (Supabase ${p.status})`}`);
  }
})().catch(e => { console.error(e.message); process.exit(1); });
```
